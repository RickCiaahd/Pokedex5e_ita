              '  Future<TeamSlot> _applyLevelUpMoves(\n'
              '    TeamSlot slot,\n'
              '    int oldLevel,\n'
              '    int newLevel,\n'
              '  ) async {\n'
              '    var selectedMoves = slot.selectedMoves.isEmpty\n'
              '        ? _defaultSelectedMoves(_pokemon, oldLevel)\n'
              '        : slot.selectedMoves.take(4).toList();\n'
              '\n'
              '    final learnedEntries =\n'
              '        _pokemon.moves.levelMoves.entries\n'
              '            .where((entry) => entry.key > oldLevel && entry.key <= newLevel)\n'
              '            .toList()\n'
              '          ..sort((a, b) => a.key.compareTo(b.key));\n'
              '\n'
              '    for (final entry in learnedEntries) {\n'
              '      for (final move in entry.value) {\n'
              '        if (selectedMoves.contains(move)) continue;\n'
              '        if (selectedMoves.length < 4) {\n'
              '          selectedMoves.add(move);\n'
              '          continue;\n'
              '        }\n'
              '\n'
              '        final moveData = _moves[move] ?? await _moveRepository.getMove(move);\n'
              '        if (!mounted) return slot.copyWith(selectedMoves: selectedMoves);\n'
              '\n'
              '        final replacedMove = await _askMoveReplacement(\n'
              '          move,\n'
              '          selectedMoves,\n'
              '          moveData,\n'
              '        );\n'
              '        if (replacedMove == null) continue;\n'
              '\n'
              '        final replaceIndex = selectedMoves.indexOf(replacedMove);\n'
              '        if (replaceIndex >= 0) selectedMoves[replaceIndex] = move;\n'
              '      }\n'
              '    }\n'
              '\n'
              '    return slot.copyWith(selectedMoves: selectedMoves);\n'
              '  }\n'

              '  Future<TeamSlot> _applyLevelUpAbilityScores(\n'
              '    TeamSlot slot,\n'
              '    int oldLevel,\n'
              '    int newLevel,\n'
              '  ) async {\n'
              '    if (_asiMilestoneCountForLevel(newLevel) <=\n'
              '        _asiMilestoneCountForLevel(oldLevel)) {\n'
              '      return slot;\n'
              '    }\n'
              '\n'
              '    var updatedSlot = slot;\n'
              '\n'
              '    while (_availableAsiPointsForSlot(updatedSlot) > 0) {\n'
              '      final remaining = _availableAsiPointsForSlot(updatedSlot);\n'
              '      final attribute = await _pickAbilityScoreIncrease(updatedSlot, remaining);\n'
              '      if (!mounted || attribute == null) break;\n'
              '\n'
              '      final currentAttributes = _attributeScores(_pokemon, updatedSlot);\n'
              '      if ((currentAttributes[attribute] ?? 0) >= 20) continue;\n'
              '\n'
              '      final customScores = Map<String, int>.from(\n'
              '        updatedSlot.customAbilityScores,\n'
              '      );\n'
              '      customScores[attribute] = (customScores[attribute] ?? 0) + 1;\n'
              '      updatedSlot = updatedSlot.copyWith(customAbilityScores: customScores);\n'
              '    }\n'
              '\n'
              '    return updatedSlot;\n'
              '  }\n'
              '\n'
              '  int _availableAsiPointsForSlot(TeamSlot slot) {\n'
              '    final level = LevelProgression.levelFromExperience(slot.experience);\n'
              '    final evolution = _evolutionForCurrentPokemon();\n'
              '    final totalStages = (evolution?.totalStages ?? 1).clamp(1, 5).toInt();\n'
              '    final pointsPerMilestone = (5 - totalStages).clamp(0, 4).toInt();\n'
              '    final earnedPoints = _asiMilestoneCountForLevel(level) * pointsPerMilestone;\n'
              '    final spentPoints = slot.customAbilityScores.values\n'
              '        .where((value) => value > 0)\n'
              '        .fold<int>(0, (sum, value) => sum + value);\n'
              '\n'
              '    return (earnedPoints - spentPoints).clamp(0, 999).toInt();\n'
              '  }\n'

              '    if (level >= 20) return 5;\n'

              '    if (level >= 8) return 2;\n'
              '    if (level >= 4) return 1;\n'
              '    return 0;\n'

              '    if (choices.isEmpty) {\n'
              '      _showMessage(\'Nessuna evoluzione disponibile per ${_pokemon.name}.\');\n'
              '      return;\n'
              '    }\n'
              '\n'
              '    final selected = await showModalBottomSheet<EvolutionEligibility>(\n'
              '      context: context,\n'
              '      showDragHandle: true,\n'
              '      isScrollControlled: true,\n'
              '      builder: (_) => EvolutionSelectorSheet(\n'
              '        currentPokemon: _pokemon,\n'
              '        choices: choices,\n'
              '        pokemonByName: _pokemonByName,\n'
              '      ),\n'
              '    );\n'
              '\n'
              '    if (!mounted || selected == null) return;\n'
              '\n'
              '    await _evolveSlot(slot, selected);\n'
              '  }\n'
              '\n'
              '  Future<TeamSlot?> _evolveSlot(\n'
              '    TeamSlot slot,\n'
              '    EvolutionEligibility selected,\n'
              '  ) async {\n'
              '    if (!selected.isAvailable) return null;\n'
              '\n'
              '    final evolvedPokemon = _pokemonForEvolutionOption(selected.option);\n'
              '    if (evolvedPokemon == null) {\n'
              '      _showMessage(\n'
              '        \'${selected.option.toName} non è presente nel catalogo attuale.\',\n'
              '      );\n'
              '      return null;\n'
              '    }\n'
              '\n'
              '    final profile = await _profileRepository.getActiveProfile();\n'
              '    final requiredItemId = selected.requiredItemId;\n'
              '    if (requiredItemId != null) {\n'
              '      final consumed = await _bagInventoryRepository.consumeItem(\n'
              '        profileId: profile.id,\n'
              '        itemId: requiredItemId,\n'
              '      );\n'
              '      if (!consumed) {\n'
              '        _showMessage(\'Oggetto evolutivo non disponibile nello zaino.\');\n'
              '        return null;\n'
              '      }\n'
              '    }\n'
              '\n'
              '    final wasFullHp = _currentHp >= _maxHp;\n'
              '    final oldName = _pokemon.name;\n'
              '    final evolvedMaxHp = _maxHpFor(evolvedPokemon, slot);\n'
              '    final updatedSlot = slot.copyWith(\n'
              '      pokemonId: evolvedPokemon.id,\n'
              '      currentHp: wasFullHp\n'
              '          ? evolvedMaxHp\n'
              '          : _currentHp.clamp(0, evolvedMaxHp).toInt(),\n'
              '      selectedMoves: List<String>.from(slot.selectedMoves),\n'
              '      formName: selected.option.targetFormName,\n'
              '    );\n'
              '\n'
              '    if (!mounted) return updatedSlot;\n'
              '\n'
              '    setState(() {\n'
              '      _basePokemon = evolvedPokemon;\n'
              '      _pokemon = evolvedPokemon;\n'
              '      _teamSlot = updatedSlot;\n'
              '      _replaceTeamSlot(updatedSlot);\n'
              '      _isLoading = true;\n'
              '      _message = \'$oldName si è evoluto in ${selected.option.toName}!\';\n'
              '    });\n'
              '\n'
              '    widget.onTeamSlotChanged?.call(updatedSlot);\n'
              '    final revealed = await _customDiscoveryService.revealByPokemonId(\n'
              '      evolvedPokemon.id,\n'
              '    );\n'
              '    if (revealed) {\n'
              '      PokemonRepository.clearCache();\n'
              '      _showMessage(\n'
              '        \'$oldName si è evoluto in ${selected.option.toName}! Nuova specie scoperta.\',\n'
              '      );\n'
              '    }\n'
              '    await _loadData();\n'
              '\n'
              '    return updatedSlot;\n'
              '  }\n'
              '\n'
              '  Future<void> _switchPartySlot(TeamSlot slot) async {\n'
              '    final pokemon = _pokemonById(slot.pokemonId);\n'
              '    if (pokemon == null) return;\n'

              '    setState(() {\n'
              '      _basePokemon = pokemon;\n'
              '      _pokemon = pokemon.resolveVariant(\n'
              '        formName: slot.formName,\n'
              '        gender: slot.gender,\n'
              '      );\n'
              '      _teamSlot = slot;\n'
              '      _isLoading = true;\n'
              '      _message = null;\n'
              '      _evolutionChoices = const [];\n'
              '    });\n'
              '    _ensureSelectedMovesIsSaved();\n'
              '    await _loadData();\n'
              '  }\n'
              '\n'
              '  int _modifier(int score) => ((score - 10) / 2).floor();\n'
              '\n'
              '  int _proficiency(int level) {\n'
              '    if (level >= 17) return 6;\n'
              '    if (level >= 13) return 5;\n'
              '    if (level >= 9) return 4;\n'
              '    if (level >= 5) return 3;\n'
              '    return 2;\n'

              '  Map<String, int> _attributeScores(Pokemon pokemon, TeamSlot? slot) {\n'

              '      pokemon: pokemon,\n'
              '      slot: slot,\n'

              '  int _bestMoveModifier(MoveData move) {\n'
              '    final attributes = _attributeScores(_pokemon, _teamSlot);\n'
              '    final modifiers = move.movePowers\n'
              '        .where(attributes.containsKey)\n'
              '        .map((power) => _modifier(attributes[power]!))\n'
              '        .toList();\n'
              '\n'
              '    if (modifiers.isEmpty) return 0;\n'
              '    modifiers.sort();\n'
              '    return modifiers.last;\n'
              '  }\n'

              '  String _moveStats(MoveData move) {\n'

              '    final proficiency = _proficiency(_level);\n'
              '    final attackPathBonus = TrainerPathPassiveService.attackRollBonus(\n'
              '      profile: _profile,\n'
              '      pokemon: _pokemon,\n'
              '      slot: _teamSlot,\n'
              '    );\n'
              '    final damagePathBonus = TrainerPathPassiveService.damageRollBonus(\n'
              '      profile: _profile,\n'
              '      slot: _teamSlot,\n'
              '    );\n'
              '    final stab = TrainerPathPassiveService.stabEffect(\n'
              '      profile: _profile,\n'
              '      pokemon: _pokemon,\n'
              '      slot: _teamSlot,\n'
              '      move: move,\n'
              '      pokemonLevel: _level,\n'
              '    );\n'
              '\n'
              '    if (move.isAttack) {\n'
              '      final attackBonus = moveModifier + proficiency + attackPathBonus;\n'
              '      parts.add(\'AB: ${attackBonus >= 0 ? \'+\' : \'\'}$attackBonus\');\n'
              '    }\n'
              '    if (move.save != null) {\n'
              '      parts.add(\'CD: ${8 + proficiency + moveModifier}\');\n'
              '    }\n'
              '\n'
              '    final damage = move.damageForLevel(_level);\n'
              '    if (damage != null) {\n'
              '      final bonus = damagePathBonus == 0\n'
              '          ? \'\'\n'
              '          : \' ${damagePathBonus > 0 ? \'+\' : \'\'}$damagePathBonus\';\n'
              '      parts.add(\'${damage.label}$bonus\');\n'
              '    }\n'
              '    if (stab.applies) {\n'
              '      final source = stab.extendedByPath ? \'STAB esteso\' : \'STAB\';\n'
              '      final bonus = stab.pathBonus == 0 ? \'\' : \' Path +${stab.pathBonus}\';\n'
              '      parts.add(\'$source$bonus\');\n'
              '    }\n'
              '    if (move.range != \'-\') parts.add(move.range);\n'
              '    if (move.duration != \'-\') parts.add(move.duration);\n'
              '\n'
              '    return parts.join(\'  ||  \');\n'
              '  }\n'

              '      padding: const EdgeInsets.all(12),\n'
